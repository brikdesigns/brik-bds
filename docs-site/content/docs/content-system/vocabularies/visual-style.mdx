  ['themeMode', 'Theme Mode'],
  ['atmosphere', 'Atmosphere'],
  ['spacingMode', 'Spacing'],

  ['radiusMode', 'Radius'],
  ['elevationMode', 'Elevation'],
  ['borderWidthMode', 'Border width'],
  ['animationTier', 'Animation tier'],
];

  subtle: 'Fade-ups on scroll only. No GSAP.',
  moderate: 'Fade-ups, a split-text hero headline, stat counters, hover lifts and an animated CSS gradient.',
  expressive: 'Split text, scroll parallax, staggered card entrances, a Granim gradient and Lenis smooth scroll.',
};

export const ATMOSPHERE_CHARACTER = {
  none: 'No atmosphere layer. Pure theme tokens.',
  'editorial-luxury': 'Quiet luxury. Blur and noise give depth; hero photography breathes.',
  'cinematic-dramatic': 'Aurora drift gradients, a centred hero glow, edge vignettes. No grain.',
  'warm-soft': 'One soft rose vignette and reduced grain.',
  'organic-textured': 'SVG grain at 40% opacity and a subtle earth-tone tint.',
};

export const PALETTE_USE = {
  Colorful: 'Uses more of the client palette on a page.',
  Monochrome: 'Uses less of the client palette on a page.',
};

export const moodsFor = (style) =>
  Object.keys(MOOD_TO_VISUAL_STYLE).filter((mood) => MOOD_TO_VISUAL_STYLE[mood].includes(style));

export const personalitiesFor = (style) => [
  ...new Set(moodsFor(style).flatMap((mood) => MOOD_TO_PERSONALITY[mood])),
];

export const blueprintsFor = (style) => {
  const moods = moodsFor(style);
  return blueprintLibrary.blueprints.filter((bp) => bp.moods.some((m) => moods.includes(m)));
};

  const preset = VISUAL_STYLE_PRESETS[style];
  return (

        </tr>
      </thead>
      <tbody>

            <td>{preset[key] ? <code>{preset[key]}</code> : 'Not set. The default applies.'}</td>
          </tr>
        ))}
      </tbody>
    </table>
  );
};

        </tr>

A Visual Style is a named look, such as Modern or Dark. A client picks up to three. This page says what each one means, and the sites and decisions behind that meaning.

The tags already exist. This page adds no new tag. It reads the preset for each style from `@brikdesigns/bds/content-system` and never copies it by hand.

## How to read a section

Each of the 13 sections has four parts.

- **Settings.** The preset, rendered from `VISUAL_STYLE_PRESETS`.
- **Four levels.** Content, theming, structure and motion. Every line names its source. A line the sources cannot support says so instead of guessing.
- **Evidence.** The sites that showed this style, with a date.
- **Decisions.** A dated log. Each row points at the evidence behind it.

A preset never sets a colour, font or logo. Those come from the client theme ([ADR-048](https://github.com/brikdesigns/brik-bds/blob/main/docs/adrs/ADR-048-visual-style-and-theming-taxonomy.md) § 1).

## Visual Style and Personality

Four words are in both lists: Bold, Minimal, Modern and Playful. Personality is the brand's character. Visual Style is how it looks. A Playful brand can use a Minimal style ([ADR-048](https://github.com/brikdesigns/brik-bds/blob/main/docs/adrs/ADR-048-visual-style-and-theming-taxonomy.md) § 4). See the [Glossary](/docs/getting-started/glossary#visual-style) for the definitions.

## How This Page Grows

- **Evidence rows** come by docs PR, from portal inspiration-site analyses ([portal#6575](https://github.com/brikdesigns/brik-client-portal/issues/6575)) and approved client homepages.
- **A decision that changes a preset** edits `VISUAL_STYLE_PRESETS` in the same PR ([ADR-048](https://github.com/brikdesigns/brik-bds/blob/main/docs/adrs/ADR-048-visual-style-and-theming-taxonomy.md) § 7).
- **Evidence that no style fits** is a proposal for a new style. Adding one is the operator's call (ADR-048 § 7).
