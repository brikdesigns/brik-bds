Prop Union Audit — 2026-11-22

Issue: #3310 (child of #3239). Every exported string-literal union in components/ui/** was checked against the written BDS rules, one value at a time, at main 520f7a37.

Agents misread the design system because the code, the docs and the gate teach three different vocabularies. The rules themselves are clear. Confidence: HIGH. The classifier reproduced all 6 of the gate's live union findings, and the docs sweep found both violations known before it ran.

  1. The gate is blind to most of it. 18 unions contradict a written rule, and lint-naming-canon sees 4 of them. Rule 5's default-deny deliberately skips variant and status unions (scripts/lint-naming-canon.mjs:906-911), and rule 3 treats them as open per-component lists (:625-626). Neither rule checks step words or snake_case. That leaves valence under status or variant, snake_case values, and retired step words in *Size / *Width unions with nothing to stop them.
  2. The known leaks have no owner. The baseline rows for Counter brand, Banner announcement and Dot/Toast default are keyed to #1925, which closed on 2026-09-04 without extracting them. A baselined finding with a closed owner reads as "accepted".
  3. The docs taught retired values as current. prop-axes.mdx listed Badge brand, Button tiny, a stale Card variant list and the retired CardPreset. prop-axes.mdx:82 and indicators.mdx:36 presented Dot's off-canon default as part of the design. Fixed in #3321.
  4. The examples break the Badge/Tag rule. 19 docs and story sites put a noun on a Badge or a status or an action on a Tag. Agents copy examples before they read rules.
  5. Deprecated names have no removal owner. 16 deprecated aliases are still exported. #1984 owns 5 of them. The rest, plus Tag onRemove and Button's retired props, had no issue until #3320.
  6. One regression. Card's retired map turns the lifecycle word failed into the valence word negative (components/ui/Card/Card.tsx:77). That is the reverse of ADR-033:131.

8. Method

  1. Inventory. The TypeScript compiler API walked every non-story, non-test .ts/.tsx file under components/ui and kept each exported type alias whose type is a union with string literals, or which points at one other type.
  2. Axis. The axis comes from the type-name suffix (…Tone → tone, …Size|Width|Tier → step, and so on). Four exceptions are named in the classifier source.
  3. Verdict. Each value is judged against tokens/naming-grammar.json and ADR-033's numbered lines.
  4. Positive control. The gate's live findings (node scripts/lint-naming-canon.mjs --no-baseline --rule 3 --json, then --rule 5; union findings BannerTone#axis, BannerTone#mixed, CounterTone#mixed, CounterTone.brand, DotTone.default, ToastTone.default) were all classified "contradict". The classifier therefore finds at least everything the gate finds.

// Inventory every exported type alias in components/ui/**/*.tsx whose type is a
// union containing string literals. Usage: node unions.cjs <repoRoot> > unions.json
const path = require('path');
const fs = require('fs');
const root = path.resolve(process.argv[2]);
const ts = require(path.join(root, 'node_modules/typescript'));

function walk(dir, out = []) {
  for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
    const p = path.join(dir, e.name);
    if (e.isDirectory()) walk(p, out);
    else if (/\.tsx?$/.test(e.name) && !/\.(stories|test|spec)\.tsx?$/.test(e.name) && !e.name.endsWith('.d.ts')) out.push(p);
  }
  return out;
}

const files = walk(path.join(root, 'components/ui'));
const rows = [];
for (const f of files) {
  const src = fs.readFileSync(f, 'utf8');
  const sf = ts.createSourceFile(f, src, ts.ScriptTarget.Latest, true, ts.ScriptKind.TSX);
  for (const st of sf.statements) {
    if (!ts.isTypeAliasDeclaration(st)) continue;
    const exported = st.modifiers?.some((m) => m.kind === ts.SyntaxKind.ExportKeyword);
    if (!exported) continue;
    const t = st.type;
    const members = ts.isUnionTypeNode(t) ? t.types : [t];
    const lits = [];
    const refs = [];
    for (const m of members) {
      if (ts.isLiteralTypeNode(m) && ts.isStringLiteral(m.literal)) lits.push(m.literal.text);
      else refs.push(m.getText(sf));
    }
    if (!lits.length) {
      // alias to another union, e.g. `export type BadgeStatus = BadgeTone`
      if (members.length === 1 && ts.isTypeReferenceNode(t)) {
        rows.push({ file: path.relative(root, f), line: sf.getLineAndCharacterOfPosition(st.getStart(sf)).line + 1, name: st.name.text, values: [], refs, alias: true });
      }
      continue;
    }
    rows.push({ file: path.relative(root, f), line: sf.getLineAndCharacterOfPosition(st.getStart(sf)).line + 1, name: st.name.text, values: lits, refs, alias: false });
  }
}
process.stdout.write(JSON.stringify(rows, null, 1));

// Per-value verdicts for #3310 AC1. Usage: node classify.cjs <repoRoot> <unions.json> <rule3.json> <rule5.json> > verdicts.json
// Verdict: match | contradict | none (no written rule). Every match/contradict carries the rule's file:line.
const path = require('path');
const [root, unionsPath, r3Path, r5Path] = process.argv.slice(2).map((p) => path.resolve(p));
const g = require(path.join(root, 'tokens/naming-grammar.json'));
const rows = require(unionsPath);
const gate = new Set([...require(r3Path).live, ...require(r5Path).live].map((f) => f.id.split(/[#.]/)[0]));

const ADR = 'docs/adrs/ADR-033-naming-canon-one-word-per-concept.md';
const PA = 'docs-site/content/docs/build-standards/prop-axes.mdx';
const R = {
  valence: `${ADR}:106-112`,
  retiredValence: `${ADR}:112`,
  defaultDeny: `${ADR}:104`,
  toneNotStatus: `${ADR}:131`,
  statusMigrate: `${ADR}:122`,
  oneAxis: `${ADR}:133`,
  brandLeak: `${ADR}:135-143`,
  snake: `${ADR}:145`,
  steps: `${ADR}:155 + ${PA}:42`,
  status: `${ADR}:123`,
  variant: `${ADR}:124`,
  curated: `${ADR}:634-640 (§ Amendments #3275)`,
  chart: 'docs/adrs/ADR-050-data-visualization-layer.md:36-38',
};
const VALENCE = new Set(g.valence.words);
const RETIRED = g.valence.retired;
const TSHIRT = new Set(g.steps.tShirt);
const RETIRED_STEPS = new Set(['standard', 'thin', 'bold', 'normal', 'fast', 'slow', 'default', 'narrow', 'wide']);
const closed = (axis) => g.axes[axis] && g.axes[axis].values;

const AXIS_BY_SUFFIX = [
  [/Tone$/, 'tone'], [/Status$/, 'status'], [/Variant$/, 'variant'], [/Emphasis$/, 'emphasis'],
  [/Appearance$/, 'appearance'], [/Density$/, 'density'], [/Orientation$/, 'orientation'],
  [/(Size|Gap|Padding|Width|Tier)$/, 'step'], [/Spacing$/, 'spacing'], [/Marker$/, 'marker'],
  [/Underline$/, 'underline'], [/Fit$/, 'fit'], [/Anchor$/, 'anchor'], [/Placement$/, 'placement'],
  [/Columns$/, 'columns'], [/Color$/, 'color'], [/Ratio$/, 'ratio'], [/Layout$/, 'layout'],
  [/Side$/, 'side'], [/Span$/, 'span'], [/ServiceLine$/, 'service-line'],
];
function axisOf(name) {
  if (name === 'ChartColor') return 'chart';
  if (name === 'ButtonVariant') return 'curated';
  // The stamp's own JSDoc says it records the control card's status (Card.tsx:83).
  if (name === 'CardControlStampKind') return 'status';
  if (name === 'TableHeaderBorderWeight') return 'step';
  for (const [re, a] of AXIS_BY_SUFFIX) if (re.test(name)) return a;
  return null;
}

function judge(u, v) {
  const axis = axisOf(u.name);
  if (/_/.test(v)) return ['contradict', R.snake, 'snake_case value; canon is kebab-case'];
  switch (axis) {
    case 'chart':
      return ['match', R.chart, 'ADR-050 chart colour set'];
    case 'curated':
      if (RETIRED[v]) return ['contradict', R.retiredValence, `retired valence word → ${RETIRED[v]}`];
      return ['match', R.curated, 'curated style list'];
    case 'tone':
      if (VALENCE.has(v)) return ['match', R.valence, 'valence word'];
      if (RETIRED[v]) return ['contradict', R.retiredValence, `retired valence word → ${RETIRED[v]}`];
      if (v === 'brand' || v === 'announcement') return ['contradict', R.brandLeak, 'brand hue in a valence union → emphasis="brand"'];
      return ['contradict', `${R.valence} + ${R.defaultDeny}`, `\`${v}\` is not one of the five valence words`];
    case 'status': {
      const others = u.values.filter((x) => x !== 'default');
      if (others.length && others.every((x) => VALENCE.has(x)))
        return ['contradict', R.statusMigrate, 'valence carried under `status` → migrate to `tone`'];
      if (VALENCE.has(v)) return ['contradict', R.toneNotStatus, 'valence word inside a lifecycle `status` union'];
      if (RETIRED[v]) return ['contradict', R.retiredValence, `retired valence word → ${RETIRED[v]}`];
      return ['match', R.status, 'per-subject lifecycle/presence word'];
    }
    case 'variant':
      if (VALENCE.has(v) && v !== 'neutral') return ['contradict', R.statusMigrate, 'valence member in `variant` (form, no valence)'];
      if (RETIRED[v]) return ['contradict', R.retiredValence, `retired valence word → ${RETIRED[v]}`];
      return ['match', R.variant, 'per-component form word'];
    case 'step':
      if (TSHIRT.has(v) || v === 'none' || v === 'full') return ['match', R.steps, 't-shirt rung / reset'];
      if (RETIRED_STEPS.has(v)) return ['contradict', R.steps, `\`${v}\` is a retired step word`];
      if (closed('density').includes(v)) return ['contradict', `${R.oneAxis}`, 'density word in a step union'];
      if (v === 'inherit') return ['none', '', 'CSS keyword; no written rule'];
      return ['contradict', `${R.steps} + ${R.defaultDeny}`, `\`${v}\` is not a t-shirt rung`];
    case 'layout': case 'ratio':
      return ['match', `${ADR} § Amendments (${axis}, open list)`, `open per-component ${axis} list`];
    default: {
      const list = axis && closed(axis);
      if (list) return list.includes(v) ? ['match', `tokens/naming-grammar.json axes.${axis}`, `closed ${axis} list`] : ['contradict', `tokens/naming-grammar.json axes.${axis}`, `not on the closed ${axis} list`];
      return ['none', '', 'no axis word governs this union'];
    }
  }
}

const out = rows.map((u) => {
  if (u.alias) return { ...u, axis: 'alias', gateSees: gate.has(u.name), verdicts: [] };
  return { ...u, axis: axisOf(u.name), gateSees: gate.has(u.name), verdicts: u.values.map((v) => { const [verdict, rule, why] = judge(u, v); return { value: v, verdict, rule, why }; }) };
});
process.stdout.write(JSON.stringify(out, null, 1));
