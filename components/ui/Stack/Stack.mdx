  <div style={{ width: w ?? 'auto', padding: 'var(--padding-md)', background: 'var(--surface-secondary)', border: '1px dashed var(--border-secondary)', borderRadius: 'var(--border-radius-sm)', fontFamily: 'var(--font-family-body)', fontSize: 'var(--body-sm)', color: 'var(--text-primary)' }}>

export const SectionLabel = ({ children }) => (
  <div style={{ fontFamily: 'var(--font-family-label)', fontSize: 'var(--body-xs)', textTransform: 'uppercase', letterSpacing: '0.05em', color: 'var(--text-muted)', marginBottom: 'var(--gap-sm)' }}>
    {children}
  </div>
);

# Stack

<ComponentLinks slug="stack" />

Stack lays out children along one axis with a consistent `gap`. Vertical by default; set `orientation="horizontal"` for a row. `align` controls cross-axis alignment.
